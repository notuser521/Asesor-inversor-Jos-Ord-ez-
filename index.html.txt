async function loadData() {
  try {
    const res = await fetch('datos.json', { cache: 'no-store' });

    globalData = await res.json();

    globalData = DATOS_EJEMPLO;
    document.getElementById('data-alert').style.display = 'block';
  }

function initQuiz() {
  currentQuestionIndex = 0;
  userScore = 0;
  showQuestion();
}

function showQuestion() {
  const q = CONFIG.preguntas[currentQuestionIndex];
  const pct = ((currentQuestionIndex + 1) / CONFIG.preguntas.length) * 100;
  document.getElementById('quiz-progress').style.width = pct + '%';
  document.getElementById('quiz-question-title').innerText = `${currentQuestionIndex + 1}. ${q.p}`;

  const container = document.getElementById('quiz-options');
  container.innerHTML = '';
  q.o.forEach(opt => {
    const btn = document.createElement('button');
    btn.className = 'btn-option';
    btn.innerText = opt.t;
    btn.onclick = () => answerQuestion(opt.pts);
    container.appendChild(btn);
  });
}

function answerQuestion(pts) {
  userScore += pts;
  currentQuestionIndex++;
  if (currentQuestionIndex < CONFIG.preguntas.length) {
    showQuestion();
  } else {
    finishQuiz();
  }
}

function finishQuiz() {
  document.getElementById('quiz-container').style.display = 'none';
  document.getElementById('profile-result-container').style.display = 'block';

  let desc = "";
  if (userScore <= 9) {
    userProfile = "Conservador";
    desc = "Su prioridad principal es proteger el capital depositado, reduciendo al mínimo el riesgo de pérdidas. Prefiere la estabilidad frente a altos rendimientos.\nPara su perfil, las acciones deberían ser una parte pequeña de su dinero; considere primero CDT, TES o fondos de renta fija.";
  } else if (userScore <= 15) {
    userProfile = "Moderado";
    desc = "Busca un equilibrio entre el crecimiento de su capital y la seguridad. Tolera variaciones moderadas en el valor de sus inversiones a cambio de rentabilidades superiores a la inflación.";
  } else {
    userProfile = "Arriesgado";
    desc = "Su objetivo principal es maximizar el crecimiento de su capital a largo plazo. Acepta una alta volatilidad y posibles caídas temporales significativas en el mercado.";
  }

  document.getElementById('profile-type').innerText = userProfile;
  document.getElementById('profile-desc').innerText = desc;

  renderStocksTable(); // Re-renderiza para mostrar las etiquetas según el perfil
}

function resetQuiz() {
  document.getElementById('quiz-container').style.display = 'block';
  document.getElementById('profile-result-container').style.display = 'none';
  initQuiz();
}

function getRiskLevel(vol) {
  if (vol < 20) return "Bajo";
  if (vol <= 35) return "Medio";
  return "Alto";
}

function renderStocksTable() {
  if (!globalData) return;
  const tbody = document.getElementById('stocks-tbody');
  tbody.innerHTML = '';

  let countFit = 0;

  globalData.acciones.forEach(st => {
    const risk = getRiskLevel(st.volatilidad_anual_pct);
    let fitsProfile = false;

    if (userProfile === 'Conservador' && risk === 'Bajo') fitsProfile = true;
    if (userProfile === 'Moderado' && (risk === 'Bajo' || risk === 'Medio')) fitsProfile = true;
    if (userProfile === 'Arriesgado') fitsProfile = true;

    if (fitsProfile) countFit++;

    const tr = document.createElement('tr');
    tr.onclick = () => openStockModal(st);

    const varDiaClass = st.variacion_dia_pct >= 0 ? 'val-pos' : 'val-neg';
    const rent1aClass = st.rent_1a_pct >= 0 ? 'val-pos' : 'val-neg';
    const rent3aClass = st.rent_anual_3a_pct >= 0 ? 'val-pos' : 'val-neg';

    tr.innerHTML = `
      <td data-label="Nombre">
        <strong>${st.nombre}</strong>
        ${fitsProfile ? '<br><span class="badge-fit">Acorde a su perfil</span>' : ''}
      </td>
      <td data-label="Nemo">${st.nemotecnico}</td>
      <td data-label="Sector">${st.sector}</td>
      <td data-label="Precio">${fmtCOP.format(st.precio)}</td>
      <td data-label="Var. Día" class="${varDiaClass}">${st.variacion_dia_pct >= 0 ? '+' : ''}${fmtPct.format(st.variacion_dia_pct / 100)}</td>
      <td data-label="Rent. 1A" class="${rent1aClass}">${st.rent_1a_pct >= 0 ? '+' : ''}${fmtPct.format(st.rent_1a_pct / 100)}</td>
      <td data-label="Rent. 3A" class="${rent3aClass}">${st.rent_anual_3a_pct >= 0 ? '+' : ''}${fmtPct.format(st.rent_anual_3a_pct / 100)}</td>
      <td data-label="Volatilidad">${fmtPct.format(st.volatilidad_anual_pct / 100)}</td>
      <td data-label="Caída Máx." class="val-neg">${fmtPct.format(st.caida_maxima_1a_pct / 100)}</td>
      <td data-label="Rend. Div.">${fmtPct.format(st.dividendo_rend_pct / 100)}</td>
      <td data-label="Riesgo">${risk}</td>
    `;
    tbody.appendChild(tr);
  });

  // Nota de pie de fuente y TRM
  let trmTxt = globalData.trm && globalData.trm.valor ? ` · TRM: ${fmtCOP.format(globalData.trm.valor)} (${globalData.trm.fecha})` : '';
  let dateTxt = globalData.actualizado ? globalData.actualizado.split('T')[0] : '';
  document.getElementById('source-info').innerText = `Fuente: ${globalData.fuente} · Datos al ${dateTxt}${trmTxt}`;

  if (userProfile && countFit === 0) {
    const alertDiv = document.createElement('div');
    alertDiv.className = 'alert-banner';
    alertDiv.style.marginTop = '10px';
    alertDiv.innerText = "Ninguna de estas acciones tiene riesgo bajo; para su perfil conviene priorizar instrumentos de renta fija.";
    tbody.parentNode.appendChild(alertDiv);
  }
}

function setSimMode(mode) {

function getSelectedStocks() {
  const type = document.getElementById('sim-asset-type').value;
  if (type === 'single') {
    const val = document.getElementById('sim-single-select').value;
    return globalData.acciones.filter(s => s.ticker === val);
  } else {
    const checked = Array.from(document.querySelectorAll('#sim-basket-list input:checked')).map(c => c.value);
    return globalData.acciones.filter(s => checked.includes(s.ticker));
  }
}

function calculateSimulation() {
  if (!globalData) return;
  const stocks = getSelectedStocks();
  if (stocks.length === 0) {
    document.getElementById('sim-main-output').innerText = "Seleccione al menos una acción.";
    return;
  }

  // Rentabilidad EA (r) y Volatilidad (sigma) promedio simple
  const rateType = document.getElementById('sim-rate-type').value;
  let r = 0;
  let sigma = 0;

  if (rateType === 'custom') {
    r = parseFloat(document.getElementById('sim-custom-rate-val').value) / 100 || 0;
    sigma = stocks.reduce((acc, s) => acc + s.volatilidad_anual_pct, 0) / stocks.length / 100;
  } else if (rateType === '1a') {
    r = stocks.reduce((acc, s) => acc + s.rent_1a_pct, 0) / stocks.length / 100;
    sigma = stocks.reduce((acc, s) => acc + s.volatilidad_anual_pct, 0) / stocks.length / 100;
  } else { // '3a' por defecto
    r = stocks.reduce((acc, s) => acc + s.rent_anual_3a_pct, 0) / stocks.length / 100;
    sigma = stocks.reduce((acc, s) => acc + s.volatilidad_anual_pct, 0) / stocks.length / 100;
  }

  // Tasa mensual efectiva
  const i = r > -1 ? Math.pow(1 + r, 1 / 12) - 1 : 0;

  // Variables de entrada
  let C = parseFloat(document.getElementById('sim-capital').value) || 0;
  let A = parseFloat(document.getElementById('sim-monthly').value) || 0;
  let n = parseFloat(document.getElementById('sim-years').value) || 0;
  let Meta = parseFloat(document.getElementById('sim-target').value) || 0;

  if (simMode === 2) C = 0;
  if (simMode === 4) A = 0;

  let VF = 0;
  let tiempoTexto = "";

  // Fórmulas principales
  if (simMode === 1) { // ¿Cuánto tendré?
    VF = C * Math.pow(1 + r, n) + (i !== 0 ? A * ((Math.pow(1 + i, 12 * n) - 1) / i) : A * 12 * n);
    document.getElementById('sim-main-output').innerText = `Valor Futuro Estimado: ${fmtCOP.format(VF)}`;
  } 
  else if (simMode === 2) { // ¿Cuánto debo invertir hoy?
    const retAportes = i !== 0 ? A * ((Math.pow(1 + i, 12 * n) - 1) / i) : A * 12 * n;
    C = (Meta - retAportes) / Math.pow(1 + r, n);
    if (C <= 0) {
      C = 0;
      document.getElementById('sim-main-output').innerText = `Capital Necesario Hoy: $ 0 (Sus aportes mensuales son suficientes para alcanzar la meta)`;
    } else {
      document.getElementById('sim-main-output').innerText = `Capital Necesario Hoy: ${fmtCOP.format(C)}`;
    }
    VF = Meta;
  } 
  else if (simMode === 3) { // ¿En cuánto tiempo llego?
    if (r <= 0 && A <= 0) {
      document.getElementById('sim-main-output').innerText = "La meta no es alcanzable sin aportes ni rentabilidad positiva.";
      return;
    }
    let m = 0;
    let acum = C;
    while (acum < Meta && m < 1200) { // máx 100 años
      m++;
      acum = acum * (1 + i) + A;
    }
    n = m / 12;
    if (m >= 1200) {
      document.getElementById('sim-main-output').innerText = "La meta requiere más de 100 años para alcanzarse.";
      return;
    } else {
      const yrs = Math.floor(m / 12);
      const mos = m % 12;
      tiempoTexto = `${yrs} años y ${mos} meses`;
      document.getElementById('sim-main-output').innerText = `Tiempo estimado: ${tiempoTexto}`;
    }
    VF = Meta;
  } 
  else if (simMode === 4) { // ¿Cuánto debo ahorrar al mes?
    const capitalFuturo = C * Math.pow(1 + r, n);
    const faltante = Meta - capitalFuturo;
    if (faltante <= 0) {
      A = 0;
      document.getElementById('sim-main-output').innerText = `Aporte Mensual Necesario: $ 0 (Su capital inicial crece hasta superar la meta)`;
    } else {
      A = i !== 0 ? (faltante * i) / (Math.pow(1 + i, 12 * n) - 1) : faltante / (12 * n);
      document.getElementById('sim-main-output').innerText = `Aporte Mensual Necesario: ${fmtCOP.format(A)}`;
    }
    VF = Meta;
  }

  // Cálculos de apoyo y comparaciones
  const totalAportado = C + (A * 12 * n);
  const ganancia = Math.max(0, VF - totalAportado);

  // Compra de acciones enteras
  let accionesInfo = "";
  const pesoPorStock = C / stocks.length;
  stocks.forEach(st => {
    const numAcc = Math.floor(pesoPorStock / st.precio);
    const sob = pesoPorStock - (numAcc * st.precio);
    accionesInfo += `<br>• <strong>${st.nemotecnico}</strong>: ${numAcc} acciones (${fmtCOP.format(numAcc * st.precio)}) — Sobrante: ${fmtCOP.format(sob)}`;
  });

  // Comparación CDT e Inflación
  const cdtRate = (parseFloat(document.getElementById('sim-cdt-rate').value) || 0) / 100;
  const infRate = (parseFloat(document.getElementById('sim-inf-rate').value) || 0) / 100;
  
  const iCDT = cdtRate > -1 ? Math.pow(1 + cdtRate, 1/12) - 1 : 0;
  const vfCDT = C * Math.pow(1 + cdtRate, n) + (iCDT !== 0 ? A * ((Math.pow(1 + iCDT, 12 * n) - 1) / iCDT) : A * 12 * n);
  const rReal = (1 + r) / (1 + infRate) - 1;

  document.getElementById('sim-details').innerHTML = `
    <strong>Total Aportado:</strong> ${fmtCOP.format(totalAportado)}<br>
    <strong>Ganancia Estimada:</strong> ${fmtCOP.format(ganancia)}<br>
    <strong>Acciones compradas hoy con capital inicial:</strong>${accionesInfo}<br><br>
    <strong>Comparación con CDT (${fmtPct.format(cdtRate)} E.A.):</strong> Obtendría ${fmtCOP.format(vfCDT)}<br>
    <strong>Rentabilidad Real (frente a inflación ${fmtPct.format(infRate)}):</strong> ${fmtPct.format(rReal)} E.A.
  `;

  // Escenarios
  const rPess = r - (sigma / Math.sqrt(Math.max(1, n)));
  const rOpt = r + (sigma / Math.sqrt(Math.max(1, n)));

  const calcVFWithR = (rate) => {
    const rateI = rate > -1 ? Math.pow(1 + rate, 1/12) - 1 : 0;
    return C * Math.pow(1 + rate, n) + (rateI !== 0 ? A * ((Math.pow(1 + rateI, 12 * n) - 1) / rateI) : A * 12 * n);
  };

  const vfPes = calcVFWithR(rPess);
  const vfOpt = calcVFWithR(rOpt);

  document.getElementById('scen-pessimistic').innerText = fmtCOP.format(vfPes);
  document.getElementById('scen-expected').innerText = fmtCOP.format(VF);
  document.getElementById('scen-optimistic').innerText = fmtCOP.format(vfOpt);

  renderSimChart(C, A, n, rPess, r, rOpt);
}

function renderSimChart(C, A, n, rPes, rExp, rOpt) {
  const labels = [];
  const dataAportes = [];
  const dataPes = [];
  const dataExp = [];
  const dataOpt = [];

  const steps = Math.min(Math.ceil(n), 30);
  const stepYears = n / steps;

  for (let k = 0; k <= steps; k++) {
    const yr = k * stepYears;
    labels.push(`Año ${yr.toFixed(1)}`);
    dataAportes.push(C + (A * 12 * yr));

    const calcVF = (rate) => {
      const rateI = rate > -1 ? Math.pow(1 + rate, 1/12) - 1 : 0;
      return C * Math.pow(1 + rate, yr) + (rateI !== 0 ? A * ((Math.pow(1 + rateI, 12 * yr) - 1) / rateI) : A * 12 * yr);
    };

    dataPes.push(calcVF(rPes));
    dataExp.push(calcVF(rExp));
    dataOpt.push(calcVF(rOpt));
  }

  const ctx = document.getElementById('chart-sim').getContext('2d');
  if (simChartInstance) simChartInstance.destroy();

  simChartInstance = new Chart(ctx, {
    type: 'line',
    data: {
      labels: labels,
      datasets: [
        { label: 'Pesimista', data: dataPes, borderColor: '#D9534F', fill: false },
        { label: 'Esperado', data: dataExp, borderColor: '#2E9E5B', fill: false, borderWidth: 3 },
        { label: 'Optimista', data: dataOpt, borderColor: '#16234F', fill: false },
        { label: 'Total Aportado', data: dataAportes, borderColor: '#888', borderDash: [5, 5], fill: false }
      ]
    },
    options: {
      responsive: true,
      maintainAspectRatio: false,
      plugins: { legend: { position: 'bottom' } },
      scales: {
        y: { ticks: { callback: value => '$' + value.toLocaleString('es-CO') } }
      }
    }
  });
}
